type flow: "Main data flow" color=#00E5FF width=3 particles=2

type alert: "Failure and cost alerts" dash="6 4" color=#FF4FB0 width=2 particles=1

group ext "External APIs" color=#C9B8FF {
  yt: YouTube Data API v3 [web] "API key"
  lfm: Last.fm API [web] "API key"
  mb: MusicBrainz API [web] "no key"
}

group aws "AWS account · free tier" color=#FFC9A8 {
  sched: EventBridge Scheduler [aws/eventbridge] "daily 07:00 Mexico City"
  group run "Daily run · Step Functions calls everything in this box, in order" color=#FFE9A8 {
    sfn: Step Functions [aws/stepfunctions] "state machine Pipeline"
    group ingest "1 · Ingestion Lambdas in parallel · Python stdlib" color=#A8D8FF {
      fyt: ingest_youtube [aws/lambda]
      flfm: ingest_lastfm [aws/lambda]
      fmb: ingest_musicbrainz [aws/lambda]
    }
    athena: Athena workgroup [aws/athena] "2 · MERGE · checks · VACUUM"
    fexp: export_site [aws/lambda] "3 · dated site JSON"
  }
  ssm: SSM Parameter Store [aws/systemsmanager] "API keys · SecureString"
  s3: S3 data bucket [aws/s3] "bronze · silver · quarantine · gold · exports"
  glue: Glue Data Catalog [aws/glue] "silver tables · projection"
  gold: Gold Iceberg tables [aws/s3] "gold_video_launch · gold_artist_daily"
  sns: SNS alert topic [aws/sns] "failed run or check"
  budget: AWS Budgets [email] "alert on the first cent"
}

group site "Static site · never calls AWS" color=#B5F2D0 {
  repo: site/data/ [storage] "latest.json + dated copy"
  pages: GitHub Pages [web] "gh-pages branch"
}

mail: Alert email [user] "maintainer"
