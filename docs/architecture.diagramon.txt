    fyt: ingest_youtube [aws/lambda]
    flfm: ingest_lastfm [aws/lambda]
    fmb: ingest_musicbrainz [aws/lambda]
  }
  ssm: SSM Parameter Store [aws/systemsmanager] "API keys · SecureString"
  s3: S3 data bucket [aws/s3] "raw · validated · quarantine"
  athena: Athena workgroup [aws/athena] "MERGE · checks · VACUUM" phase=p2
  glue: Glue Data Catalog [aws/glue] "silver tables · projection" phase=p2
  gold: Gold Iceberg tables [aws/s3] "mv_launch · artist_daily" phase=p2
  sns: SNS alert topic [aws/sns] "failed run or check"
  budget: AWS Budgets [email] "alert on the first cent"
}

group site "No AWS connection · planned" color=#8b93a1 phase=p3 {
  export: Dated export [storage] "site/data/"
  pages: GitHub Pages [web] "static site"
}

mail: Alert email [user] "maintainer"
