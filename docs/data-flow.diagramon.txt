group ext "External APIs" color=#C9B8FF {
  yt: YouTube Data API v3 [web] "video counters"
  lfm: Last.fm API [web] "artist and track plays"
  mb: MusicBrainz API [web] "release groups"
}

ingest: Ingestion Lambdas [aws/lambda] "check the data contracts"

group bucket "S3 data bucket · one folder per snapshot_date" color=#FFC9A8 {
  raw: raw/ [aws/s3] layer=bronze "API responses, untouched"
  validated: validated/ [aws/s3] layer=silver "passed the contract"
  quarantine: quarantine/ [aws/s3] "failed, with the reasons"
  gold: gold/ · Iceberg [aws/s3] layer=gold "video_launch · artist_daily"
  exports: exports/ [aws/s3] "dated site.json + latest"
}

silver: Silver tables [aws/glue] layer=silver "YouTube · Last.fm"
export: export_site [aws/lambda] "after the gold checks pass"

group site "Static site · never calls AWS" color=#B5F2D0 {
  repo: site/data/ [storage] "latest.json + dated copy"
  pages: GitHub Pages [web] "charts and comparisons"
}

yt -> ingest : HTTPS style=flow

ingest -> raw : as returned style=flow
ingest -> validated : passes style=flow
ingest -> quarantine : fails · over 5% stops the run style=reject
validated -> silver : Glue catalog style=flow
silver -> gold : Athena MERGE style=flow

export -> exports : JSON style=flow
